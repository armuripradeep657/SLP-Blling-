   433→   433→/* Changes apply instantly so the owner sees the look right away.       */
   434→   434→/* ------------------------------------------------------------------ */
   435→   435→
   436→   436→function AppearanceSection() {
   437→   437→  const t = useT();
   438→   438→  const appLang = useAppLang();
   439→   439→  const appearance = useStore((s) => s.appearance);
   440→   440→  const setAppearance = useStore((s) => s.setAppearance);
   441→   441→  const { toast } = useToast();
   442→   442→  const fileRef = useRef<HTMLInputElement>(null);
   443→   443→
   444→   444→  /** Downscale the picked photo so it stays small enough for local storage */
   445→   445→  const pickWatermark = async (e: React.ChangeEvent<HTMLInputElement>) => {
   446→   446→    const f = e.target.files?.[0];
   447→   447→    e.target.value = '';
   448→   448→    if (!f) return;
   449→   449→    try {
   450→   450→      const url = URL.createObjectURL(f);
   451→   451→      const img = await new Promise<HTMLImageElement>((res, rej) => {
   452→   452→        const im = new Image();
   453→   453→        im.onload = () => res(im);
   454→   454→        im.onerror = () => rej(new Error('bad image'));
   455→   455→        im.src = url;
   456→   456→      });
   457→   457→      const max = 640;
   458→   458→      const scale = Math.min(1, max / Math.max(img.naturalWidth, img.naturalHeight));
   459→   459→      const w = Math.max(1, Math.round(img.naturalWidth * scale));
   460→   460→      const h = Math.max(1, Math.round(img.naturalHeight * scale));
   461→   461→      const canvas = document.createElement('canvas');
   462→   462→      canvas.width = w;

   705→   705→/**
   706→   706→ * Every device runs the SAME app link — when the owner (or we) change anything,
   707→   707→ * installed phones pick it up automatically: the app checks for updates while
   708→   708→ * open (and on every open/focus/back-online) and refreshes itself. This button
   709→   709→ * lets the owner force the check right now.
   710→   710→ */
   711→   711→function AppUpdatesSection() {
   712→   712→  const { toast } = useToast();
   713→   713→  const [checking, setChecking] = useState(false);
   714→   714→
   715→   715→  const checkNow = async () => {
   716→   716→    if (!('serviceWorker' in navigator)) {
   717→   717→      toast({ title: 'You are on the latest version' });
   718→   718→      return;
   719→   719→    }
   720→   720→    setChecking(true);
   721→   721→    try {
   722→   722→      const reg = await navigator.serviceWorker.getRegistration();
   723→   723→      if (!reg) {
   724→   724→        toast({ title: 'You are on the latest version' });
   725→   725→        return;
   726→   726→      }
   727→   727→      let found = false;
   728→   728→      const onUpdateFound = () => {
   729→   729→        found = true;
   730→   730→      };
   731→   731→      reg.addEventListener('updatefound', onUpdateFound, { once: true });
   732→   732→      await reg.update();
   733→   733→      // give a freshly found worker a moment to install (skipWaiting applies it)
   734→   734→      await new Promise((r) => setTimeout(r, 1200));
   735→   735→      reg.removeEventListener('updatefound', onUpdateFound);
   736→   736→      if (found || reg.waiting || reg.installing) {
   737→   737→        toast({
   738→   738→          title: 'Update found',
   739→   739→          description: 'The app refreshes to the latest version in a moment…',
   740→   740→        });
   741→   741→      } else {
   742→   742→        toast({ title: 'You are on the latest version' });
   743→   743→      }
   744→   744→    } catch {
   745→   745→      toast({
   746→   746→        title: 'Cannot check right now',
   747→   747→        description: 'Go online once and tap again — the app will update itself.',
   748→   748→      });
   749→   749→    } finally {
   750→   750→      setChecking(false);
   751→   751→    }
   752→   752→  };
   753→   753→
   754→   754→  return (
   755→   755→    <section aria-label="App updates" className="space-y-2">
   756→   756→      <h3 className="text-sm font-bold text-blue-800">App updates</h3>
   757→   757→      <p className="text-xs text-muted-foreground">
   758→   758→        The same app link works on every phone. When anything is updated here, all installed
   759→   759→        devices get the change automatically — no reinstall needed. Bills stay safe on each
   760→   760→        device.
   761→   761→      </p>
   762→   762→      <Button variant="outline" className="w-full" onClick={checkNow} disabled={checking}>
   763→   763→        {checking ? <RefreshCw className="size-4 animate-spin" /> : <RefreshCw className="size-4" />}
   764→   764→        Check for updates now
   765→   765→      </Button>
   766→   766→    </section>
   767→   767→  );
   768→   768→}
