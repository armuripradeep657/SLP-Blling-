    52→    52→type LedgerTab = 'day' | 'allDays' | 'income' | 'expenses' | 'dues';
    53→    53→
    54→    54→/** One money row in the ledger — income (bill payment) or spent (expense) */
    55→    55→interface LedgerRow {
    56→    56→  time: string;
    57→    57→  /** raw timestamp — sort key so rows stay chronological across am/pm */
    58→    58→  ts: number;
    59→    59→  /** main line — customer name for income, category for spent */
    60→    60→  label: string;
    61→    61→  /** secondary line — invoice no / note */
    62→    62→  sub: string;
    63→    63→  amount: number;
    64→    64→  kind: 'in' | 'out';
    65→    65→  /** Cash / UPI / Card — shown as a chip on every row */
    66→    66→  mode?: PaymentMode;
    67→    67→}
    68→    68→
    69→    69→function categoryLabel(c: ExpenseCategory, appLang: AppLanguage): string {
    70→    70→  return appLang === 'te' ? CATEGORY_TE[c] : c;
    71→    71→}
    72→    72→
    73→    73→function modeLabel(m: PaymentMode, appLang: AppLanguage): string {
    74→    74→  if (appLang !== 'te') return m;
    75→    75→  return m === 'Cash' ? 'నగదు' : m === 'Card' ? 'కార్డ్' : m === 'Due' ? 'బాకీ' : 'UPI';
    76→    76→}
    77→    77→
    78→    78→/** Cash / UPI / Card chip — on income and spent rows alike */
    79→    79→function ModeChip({ mode, appLang }: { mode?: PaymentMode; appLang: AppLanguage }) {
    80→    80→  if (!mode) return null;
    81→    81→  const cls =
    82→    82→    mode === 'Cash'
    83→    83→      ? 'bg-green-100 text-green-800'
    84→    84→      : mode === 'UPI'
    85→    85→        ? 'bg-blue-100 text-blue-800'
    86→    86→        : mode === 'Due'
    87→    87→          ? 'bg-amber-100 text-amber-800'
    88→    88→          : 'bg-purple-100 text-purple-800';
    89→    89→  return (
    90→    90→    <span className={`inline-flex shrink-0 items-center rounded-full px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wide ${cls}`}>
    91→    91→      {modeLabel(mode, appLang)}
    92→    92→    </span>
    93→    93→  );
    94→    94→}
    95→    95→
    96→    96→/** Stat card used across ledger tabs (module-level — never created during render) */
    97→    97→function StatCard({ label, value, kind }: { label: string; value: number; kind: 'in' | 'out' | 'net' }) {
    98→    98→  return (
    99→    99→    <div className="anim-pop rounded-xl border bg-card p-3">
   100→   100→      <p className="flex items-center gap-1 text-xs text-muted-foreground">
   101→   101→        {kind === 'in' ? (
   102→   102→          <ArrowUpRight className="size-3.5 text-green-700" />
   103→   103→        ) : kind === 'out' ? (
   104→   104→          <ArrowDownRight className="size-3.5 text-red-600" />
   105→   105→        ) : (
   106→   106→          <TrendingUp className="size-3.5 text-blue-700" />
   107→   107→        )}
   108→   108→        {label}
   109→   109→      </p>
   110→   110→      <p
   111→   111→        className={`mt-0.5 text-lg font-extrabold tabular-nums ${
   112→   112→          kind === 'in' ? 'text-green-700' : kind === 'out' ? 'text-red-600' : value < 0 ? 'text-red-600' : 'text-blue-800'
   113→   113→        }`}
   114→   114→      >
   115→   115→        ₹{fmt(value)}
   116→   116→      </p>
   117→   117→    </div>
   118→   118→  );
   119→   119→}

   154→   154→}

   156→   156→export default function LedgerView() {
   157→   157→  const t = useT();
   158→   158→  const appLang = useAppLang();
   159→   159→  const invoices = useStore((s) => s.invoices);
   160→   160→  const expenses = useStore((s) => s.expenses);
   161→   161→  const incomes = useStore((s) => s.incomes);
   162→   162→  const receiveDuePayment = useStore((s) => s.receiveDuePayment);
   163→   163→  const addExpense = useStore((s) => s.addExpense);
   164→   164→  const deleteExpense = useStore((s) => s.deleteExpense);
   165→   165→  const addIncome = useStore((s) => s.addIncome);
   166→   166→  const deleteIncome = useStore((s) => s.deleteIncome);
   167→   167→  const { toast } = useToast();
   168→   168→
   169→   169→  const today = localDateKey(new Date());
   170→   170→  const thisMonth = localMonthKey(new Date());
   171→   171→
   172→   172→  const [tab, setTab] = useState<LedgerTab>('day');
   173→   173→  const [day, setDay] = useState(today);
   174→   174→  const [month, setMonth] = useState(thisMonth);
   175→   175→  const [expandedDay, setExpandedDay] = useState<string | null>(null);
   176→   176→
   177→   177→  /* ---------------- Money aggregation ---------------- */
   178→   178→
   179→   179→  // Every income / spent entry grouped by local day — feeds the day view,
   180→   180→  // the all-days list AND each day's expandable detail (name + Cash/UPI)
   181→   181→  const rowsByDay = useMemo(() => {
   182→   182→    const map = new Map<string, LedgerRow[]>();
   183→   183→    const push = (k: string, r: LedgerRow) => {
   184→   184→      const list = map.get(k);
   185→   185→      if (list) list.push(r);
   186→   186→      else map.set(k, [r]);
   187→   187→    };
   188→   188→    invoices.forEach((inv) => {
   189→   189→      paymentsOf(inv).forEach((p) => {
   190→   190→        const d = new Date(p.dateISO);
   191→   191→        push(localDateKey(d), {
   192→   192→          time: fmtTime(d),
   193→   193→          ts: d.getTime(),
   194→   194→          label: `${t('ledger.receivedFrom')} ${inv.customer.name || inv.no}`,
   195→   195→          sub: inv.no,
   196→   196→          amount: p.amount,
   197→   197→          kind: 'in',
   198→   198→          mode: p.mode,
   199→   199→        });
   200→   200→      });
   201→   201→    });
   202→   202→    expenses.forEach((e) => {
   203→   203→      const d = new Date(e.dateISO);
   204→   204→      push(localDateKey(d), {
   205→   205→        time: fmtTime(d),
   206→   206→        ts: d.getTime(),
   207→   207→        // person is the row name when given — else the category
   208→   208→        label: e.person ? e.person : categoryLabel(e.category, appLang),
   209→   209→        sub: [e.person ? categoryLabel(e.category, appLang) : '', e.note || '']
   210→   210→          .filter(Boolean)
   211→   211→          .join(' · '),
   212→   212→        amount: e.amount,
   213→   213→        kind: 'out',
   214→   214→        mode: e.mode,
   215→   215→      });
   216→   216→    });
   217→   217→    // manual income entries — money received outside a bill, with its own name
   218→   218→    incomes.forEach((e) => {
   219→   219→      const d = new Date(e.dateISO);
   220→   220→      push(localDateKey(d), {
   221→   221→        time: fmtTime(d),
   222→   222→        ts: d.getTime(),
   223→   223→        label: e.name,
   224→   224→        sub: '',
   225→   225→        amount: e.amount,
   226→   226→        kind: 'in',
   227→   227→        mode: e.mode,
   228→   228→      });
   229→   229→    });
   230→   230→    // newest first within each day (by real timestamp — safe across am/pm)
   231→   231→    map.forEach((list, k) => map.set(k, [...list].sort((a, b) => b.ts - a.ts)));
   232→   232→    return map;
   233→   233→  }, [invoices, expenses, incomes, t, appLang]);
   234→   234→

   237→   237→  const moneyByDay = useMemo(() => {
   238→   238→    const map = new Map<string, { income: number; spent: number }>();
   239→   239→    rowsByDay.forEach((rows, k) => {
   240→   240→      const acc = { income: 0, spent: 0 };
   241→   241→      rows.forEach((r) => (r.kind === 'in' ? (acc.income += r.amount) : (acc.spent += r.amount)));
   242→   242→      map.set(k, { income: Math.round(acc.income * 100) / 100, spent: Math.round(acc.spent * 100) / 100 });
   243→   243→    });
   244→   244→    return map;
   245→   245→  }, [rowsByDay]);
   246→   246→
   247→   247→  const dayMoney = (k: string): { income: number; spent: number } =>
   248→   248→    moneyByDay.get(k) ?? { income: 0, spent: 0 };
   249→   249→
   250→   250→  const dues = useMemo(
   251→   251→    () => invoices.filter((i) => i.totals.balance > 0).sort((a, b) => b.dateISO.localeCompare(a.dateISO)),
   252→   252→    [invoices]
   253→   253→  );
   254→   254→  const totalDue = dues.reduce((s, i) => s + i.totals.balance, 0);
   255→   255→
   256→   256→  // all-days rows for the selected month
   257→   257→  const monthDays = useMemo(
   258→   258→    () =>
   259→   259→      [...moneyByDay.keys()]
   260→   260→        .filter((k) => localMonthKey(new Date(`${k}T12:00:00`)) === month)
   261→   261→        .sort((a, b) => b.localeCompare(a))
   262→   262→        .map((k) => ({ key: k, ...(moneyByDay.get(k) as { income: number; spent: number }) })),
   263→   263→    [moneyByDay, month]
   264→   264→  );
   265→   265→
   266→   266→  const monthIncome = monthDays.reduce((s, d) => s + d.income, 0);
   267→   267→  const monthSpent = monthDays.reduce((s, d) => s + d.spent, 0);
   268→   268→
   269→   269→  const monthExpenses = useMemo(
   270→   270→    () =>
   271→   271→      expenses
   272→   272→        .filter((e) => localMonthKey(new Date(e.dateISO)) === month)
   273→   273→        .sort((a, b) => b.dateISO.localeCompare(a.dateISO)),
   274→   274→    [expenses, month]
   275→   275→  );
   276→   276→
   277→   277→  const monthIncomes = useMemo(
   278→   278→    () =>
   279→   279→      incomes
   280→   280→        .filter((e) => localMonthKey(new Date(e.dateISO)) === month)
   281→   281→        .sort((a, b) => b.dateISO.localeCompare(a.dateISO)),
   282→   282→    [incomes, month]
   283→   283→  );
   284→   284→
   285→   285→  /* ---------------- Receive-due inline panel ---------------- */
   286→   286→  const [receiving, setReceiving] = useState<string | null>(null);
   287→   287→  const [recvAmount, setRecvAmount] = useState('');
   288→   288→  const [recvMode, setRecvMode] = useState<PaymentMode>('Cash');
   289→   289→
   290→   290→  const openReceive = (id: string, balance: number) => {
   291→   291→    setReceiving(id);
   292→   292→    setRecvAmount(String(fmtShort(balance)));
   293→   293→    setRecvMode('Cash');
   294→   294→  };
   295→   295→
   296→   296→  const confirmReceive = () => {
   297→   297→    if (!receiving) return;
   298→   298→    const amt = Number(recvAmount) || 0;
   299→   299→    if (amt <= 0) return;
   300→   300→    receiveDuePayment(receiving, amt, recvMode);
   301→   301→    setReceiving(null);
   302→   302→    toast({ title: t('ledger.receiveDone'), description: `₹${fmt(amt)} · ${modeLabel(recvMode, appLang)}` });
   303→   303→  };
   304→   304→
   305→   305→  /* ---------------- Expense form ---------------- */
   306→   306→  const [expDate, setExpDate] = useState(today);
   307→   307→  const [expTime, setExpTime] = useState(() => nowHM());
   308→   308→  const [expCat, setExpCat] = useState<ExpenseCategory>('Labour wages');
   309→   309→  const [expPerson, setExpPerson] = useState('');
   310→   310→  const [expAmount, setExpAmount] = useState('');
   311→   311→  const [expPurpose, setExpPurpose] = useState('');
   312→   312→  const [expMode, setExpMode] = useState<PaymentMode>('Cash');

   328→   328→      category: expCat,
   329→   329→      amount: amt,
   330→   330→      person: person || undefined,
   331→   331→      note: purpose || undefined,
   332→   332→      mode: expMode,
   333→   333→    });
   334→   334→    setExpPerson('');
   335→   335→    setExpAmount('');
   336→   336→    setExpPurpose('');
   337→   337→    setExpTime(nowHM());
   338→   338→    toast({
   339→   339→      title: t('ledger.expAdded'),
   340→   340→      description: `${person ? `${person} · ` : ''}₹${fmt(amt)} · ${categoryLabel(expCat, appLang)} · ${modeLabel(expMode, appLang)}`,
   341→   341→    });
   342→   342→  };
   343→   343→
   344→   344→  /* ---------------- Manual income form ---------------- */
   345→   345→  const [incDate, setIncDate] = useState(today);
   346→   346→  const [incTime, setIncTime] = useState(() => nowHM());
   347→   347→  const [incName, setIncName] = useState('');
   348→   348→  const [incAmount, setIncAmount] = useState('');

   350→   350→
   351→   351→  const submitIncome = () => {

   356→   356→      name: incName.trim(),
   357→   357→      amount: amt,
   358→   358→      mode: incMode,
   359→   359→    });
   360→   360→    setIncName('');
   361→   361→    setIncAmount('');
   362→   362→    setIncTime(nowHM());
   363→   363→    toast({
   364→   364→      title: t('ledger.incAdded'),
   365→   365→      description: `₹${fmt(amt)} · ${incName.trim()} · ${modeLabel(incMode, appLang)}`,
   366→   366→    });
   367→   367→  };
   368→   368→
   369→   369→  const tabs: { id: LedgerTab; label: string }[] = [
   370→   370→    { id: 'day', label: t('ledger.day') },
   371→   371→    { id: 'allDays', label: t('ledger.allDays') },
   372→   372→    { id: 'income', label: t('ledger.income') },
   373→   373→    { id: 'expenses', label: t('ledger.expenses') },
   374→   374→    { id: 'dues', label: `${t('ledger.dues')}${totalDue > 0 ? ` (${dues.length})` : ''}` },
   375→   375→  ];
   376→   376→
   377→   377→  return (
   378→   378→    <div className="space-y-3">
   379→   379→      {/* Tab chips */}
   380→   380→      <div className="flex gap-1.5 overflow-x-auto pb-1" role="tablist" aria-label="Ledger sections">

   383→   383→            key={x.id}
   384→   384→            role="tab"
   385→   385→            aria-selected={tab === x.id}
   386→   386→            onClick={() => setTab(x.id)}
   387→   387→            className={`shrink-0 rounded-full px-3.5 py-1.5 text-xs font-medium transition-all active:scale-95 ${

   389→   389→            }`}
   390→   390→          >
   391→   391→            {x.label}
   392→   392→          </button>
   393→   393→        ))}
   394→   394→      </div>
   395→   395→
   396→   396→      {/* ---------- DAY VIEW ---------- */}
   397→   397→      {tab === 'day' && (
   398→   398→        <div className="space-y-3">
   399→   399→          <div className="flex items-center gap-2">
   400→   400→            <CalendarDays className="size-4 shrink-0 text-blue-700" />

   406→   406→            )}
   407→   407→          </div>
   408→   408→          <div className="grid grid-cols-3 gap-2">

   410→   410→            <StatCard label={t('ledger.spent')} value={dayMoney(day).spent} kind="out" />
   411→   411→            <StatCard label={t('ledger.net')} value={dayMoney(day).income - dayMoney(day).spent} kind="net" />
   412→   412→          </div>
   413→   413→          <ul className="space-y-1.5">
   414→   414→            {dayRows.map((r, i) => (
   415→   415→              <LedgerRowItem key={`${r.time}-${r.label}-${i}`} r={r} appLang={appLang} i={i} />
   416→   416→            ))}
   417→   417→            {dayRows.length === 0 && (
   418→   418→              <li className="py-10 text-center text-sm text-muted-foreground">{t('ledger.noActivity')}</li>
   419→   419→            )}
   420→   420→          </ul>
   421→   421→        </div>
   422→   422→      )}
