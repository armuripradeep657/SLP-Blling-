   145→   145→/** Bilingual labels — Telugu first, English second (used when bill language = Both) */
   146→   146→export const BOTH_LABELS: SlipLabels = (() => {
   147→   147→  const out = {} as SlipLabels;
   148→   148→  (Object.keys(EN_LABELS) as (keyof SlipLabels)[]).forEach((k) => {
   149→   149→    const en = EN_LABELS[k];
   150→   150→    const te = TE_LABELS[k];
   151→   151→    if (k === 'modes') {
   152→   152→      out.modes = { Cash: 'నగదు / Cash', UPI: 'UPI', Card: 'కార్డ్ / Card', Due: 'బాకీ / Due' };
   153→   153→    } else if (typeof en === 'string' && typeof te === 'string') {
   154→   154→      (out as unknown as Record<string, string>)[k] = `${te} / ${en}`;
   155→   155→    } else {
   156→   156→      (out as unknown as Record<string, unknown>)[k] = en;
   157→   157→    }
   158→   158→  });
   159→   159→  return out;
   160→   160→})();

   164→   164→}

   167→   167→export function cur(lang: BillLanguage | undefined): string {

   171→   171→/** Company name in the bill language — Telugu name when set, English otherwise */
   172→   172→export function companyNameFor(
   173→   173→  s: { companyName: string; companyNameTe?: string },
   174→   174→  lang: BillLanguage | undefined
   175→   175→): string {
   176→   176→  return lang === 'te' && s.companyNameTe && s.companyNameTe.trim() ? s.companyNameTe : s.companyName;
   177→   177→}
   178→   178→
   179→   179→/**
   180→   180→ * Company name lines for printed documents — "Both" prints the Telugu name
   181→   181→ * above the English name.
   182→   182→ */
   183→   183→export function companyLines(
   184→   184→  s: { companyName: string; companyNameTe?: string },
   185→   185→  lang: BillLanguage | undefined
   186→   186→): string[] {
   187→   187→  const te = (s.companyNameTe ?? '').trim();
   188→   188→  const en = (s.companyName ?? '').trim();
   189→   189→  if (lang === 'te') return [te || en];
   190→   190→  if (lang === 'both') {
   191→   191→    const lines = te && te !== en ? [te, en] : [en || te];
   192→   192→    return lines.filter(Boolean);
   193→   193→  }
   194→   194→  return [en || te];
   195→   195→}
   196→   196→
   197→   197→/** Item name in the bill language — Telugu product name when set, otherwise the English one */
   198→   198→export function itemLabel(it: { name: string; nameTe?: string }, lang: BillLanguage | undefined): string {
   199→   199→  const te = (it.nameTe ?? '').trim();
   200→   200→  if (lang === 'te') return te || it.name;
   201→   201→  if (lang === 'both') return te && te !== it.name ? `${te} / ${it.name}` : it.name;
   202→   202→  return it.name;
   203→   203→}
   204→   204→
   205→   205→/** Fill {tokens} in a label template */
   206→   206→export function fill(template: string, vars: Record<string, string>): string {
   207→   207→  return template.replace(/\{(\w+)\}/g, (_, k) => vars[k] ?? '');
   208→   208→}
