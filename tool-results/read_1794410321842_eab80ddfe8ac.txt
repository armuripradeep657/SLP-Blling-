   433→/* Changes apply instantly so the owner sees the look right away.       */

   435→
   436→function AppearanceSection() {
   437→  const t = useT();
   438→  const appLang = useAppLang();
   439→  const appearance = useStore((s) => s.appearance);
   440→  const setAppearance = useStore((s) => s.setAppearance);
   441→  const { toast } = useToast();
   442→  const fileRef = useRef<HTMLInputElement>(null);
   443→
   444→  /** Downscale the picked photo so it stays small enough for local storage */
   445→  const pickWatermark = async (e: React.ChangeEvent<HTMLInputElement>) => {
   446→    const f = e.target.files?.[0];
   447→    e.target.value = '';
   448→    if (!f) return;
   449→    try {
   450→      const url = URL.createObjectURL(f);
   451→      const img = await new Promise<HTMLImageElement>((res, rej) => {
   452→        const im = new Image();
   453→        im.onload = () => res(im);
   454→        im.onerror = () => rej(new Error('bad image'));
   455→        im.src = url;
   456→      });
   457→      const max = 640;
   458→      const scale = Math.min(1, max / Math.max(img.naturalWidth, img.naturalHeight));
   459→      const w = Math.max(1, Math.round(img.naturalWidth * scale));
   460→      const h = Math.max(1, Math.round(img.naturalHeight * scale));
   461→      const canvas = document.createElement('canvas');
   462→      canvas.width = w;
   463→      canvas.height = h;
   464→      canvas.getContext('2d')?.drawImage(img, 0, 0, w, h);
   465→      const dataUrl = canvas.toDataURL('image/jpeg', 0.82);
   466→      URL.revokeObjectURL(url);
   467→      setAppearance({ watermark: dataUrl });
   468→      toast({ title: t('settings.watermarkDone') });
   469→    } catch {
   470→      toast({ title: t('settings.watermarkFailed') });
   471→    }
   472→  };
   473→
   474→  return (
   475→    <section aria-label="Appearance" className="space-y-3">
   476→      <h3 className="text-sm font-bold text-blue-800">{t('settings.appearance')}</h3>
   477→      <p className="text-xs text-muted-foreground">{t('settings.appearanceDesc')}</p>
   478→
   479→      {/* Background theme */}
   480→      <div className="space-y-1.5">
   481→        <Label className="text-xs text-muted-foreground">{t('settings.bgTheme')}</Label>
   482→        <div className="grid grid-cols-5 gap-1.5">
   483→          {BG_THEMES.map((b) => (
   484→            <button
   485→              key={b.id}
   486→              type="button"
   487→              onClick={() => setAppearance({ bgTheme: b.id })}
   488→              aria-label={`Background theme ${b.id}`}
   489→              aria-pressed={appearance.bgTheme === b.id}
   490→              className={`flex flex-col items-center gap-1 rounded-lg border p-1.5 transition-all active:scale-95 ${
   491→                appearance.bgTheme === b.id ? 'border-blue-600 ring-2 ring-blue-600/30' : 'border-border'
   492→              }`}
   493→            >
   494→              <span className={`h-8 w-full rounded bg-gradient-to-b ${b.swatch}`} />
   495→              <span className="text-[10px] font-medium">{appLang === 'te' ? b.te : b.en}</span>
   496→            </button>
   497→          ))}
   498→        </div>
   499→      </div>
   500→
   501→      {/* Header colour */}
   502→      <div className="space-y-1.5">
   503→        <Label className="text-xs text-muted-foreground">{t('settings.headerColour')}</Label>
   504→        <div className="grid grid-cols-5 gap-1.5">
   505→          {HEADER_COLOURS.map((c) => (
   506→            <button
   507→              key={c.id}
   508→              type="button"
   509→              onClick={() => setAppearance({ headerTheme: c.id })}
   510→              aria-label={`Header colour ${c.id}`}
   511→              aria-pressed={appearance.headerTheme === c.id}
   512→              className={`flex flex-col items-center gap-1 rounded-lg border p-1.5 transition-all active:scale-95 ${
   513→                appearance.headerTheme === c.id ? 'border-blue-600 ring-2 ring-blue-600/30' : 'border-border'
   514→              }`}
   515→            >
   516→              <span className={`h-8 w-full rounded bg-gradient-to-r ${c.swatch}`} />
   517→              <span className="text-[10px] font-medium">{appLang === 'te' ? c.te : c.en}</span>
   518→            </button>
   519→          ))}
   520→        </div>
   521→      </div>
   522→
   523→      {/* Watermark photo from the owner's device */}
   524→      <div className="space-y-1.5">
   525→        <Label className="text-xs text-muted-foreground">{t('settings.watermark')}</Label>
   526→        <p className="text-xs text-muted-foreground">{t('settings.watermarkDesc')}</p>
   527→        <div className="flex items-center gap-3">
   528→          {/* eslint-disable-next-line @next/next/no-img-element */}
   529→          <img
   530→            src={appearance.watermark || '/narasimha-logo.jpg'}
   531→            alt="Watermark preview"
   532→            className="size-14 shrink-0 rounded-full border object-cover"
   533→            data-watermark-preview
   534→          />
   535→          <div className="flex min-w-0 flex-1 flex-col gap-1.5">
   536→            <Button variant="outline" className="h-10 w-full" onClick={() => fileRef.current?.click()} aria-label="Pick watermark photo">
   537→              <ImageUp className="size-4" />

   565→
   566→function BackupSection() {
   567→  const t = useT();
   568→  const settings = useStore((s) => s.settings);
   569→  const products = useStore((s) => s.products);
   570→  const invoices = useStore((s) => s.invoices);
   571→  const expenses = useStore((s) => s.expenses);
   572→  const incomes = useStore((s) => s.incomes);
   573→  const labourWorks = useStore((s) => s.labourWorks);
   574→  const appearance = useStore((s) => s.appearance);
   575→  const saveSettings = useStore((s) => s.saveSettings);
   576→  const { toast } = useToast();
   577→  const fileRef = useRef<HTMLInputElement>(null);
   578→  const [pending, setPending] = useState<BackupFile | null>(null);
   579→
   580→  const doBackup = () => {
   581→    const data: BackupFile = {
   582→      app: 'slp-billing',
   583→      backupVersion: 2,
   584→      exportedAt: new Date().toISOString(),
   585→      settings,
   586→      products,
   587→      invoices,
   588→      expenses,
   589→      incomes,
   590→      labourWorks,
   591→      appearance,
   592→    };
   593→    try {
   594→      const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
   595→      const url = URL.createObjectURL(blob);
   596→      const a = document.createElement('a');
   597→      a.href = url;
   598→      a.download = `slp-backup-${new Date().toISOString().slice(0, 10)}.json`;
   599→      document.body.appendChild(a);
   600→      a.click();
   601→      a.remove();
   602→      setTimeout(() => URL.revokeObjectURL(url), 5000);
   603→      toast({
   604→        title: t('settings.backupDone'),
   605→        description: `${products.length} ${t('settings.backupProducts')} · ${invoices.length} ${t('settings.backupBills')}`,
   606→      });
   607→    } catch {
   608→      toast({ title: t('settings.backupFailed') });
   609→    }
   610→  };
   611→
   612→  const onFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
   613→    const f = e.target.files?.[0];
   614→    e.target.value = '';
   615→    if (!f) return;
   616→    try {
   617→      const data = JSON.parse(await f.text()) as BackupFile;
   618→      if (data?.app !== 'slp-billing' || typeof data.settings !== 'object' || !Array.isArray(data.products)) {
   619→        throw new Error('not a valid backup');
   620→      }
   621→      setPending(data);
   622→    } catch {
   623→      toast({ title: t('settings.backupInvalid') });
   624→    }
   625→  };
   626→
   627→  const doRestore = () => {
   628→    if (!pending) return;
   629→    saveSettings(pending.settings);
   630→    useStore.setState({
   631→      products: pending.products,
   632→      invoices: pending.invoices ?? [],
   633→      expenses: pending.expenses ?? [],
   634→      incomes: pending.incomes ?? [],
   635→      labourWorks: pending.labourWorks ?? [],
   636→      ...(pending.appearance ? { appearance: { ...useStore.getState().appearance, ...pending.appearance } } : {}),
   637→    });
   638→    toast({
   639→      title: t('settings.restoredToast'),
   640→      description: `${pending.products.length} ${t('settings.backupProducts')} · ${(pending.invoices ?? []).length} ${t('settings.backupBills')}`,
   641→    });
   642→    setPending(null);
   643→  };
   644→
   645→  return (
   646→    <section aria-label="Backup and restore" className="space-y-2">
   647→      <h3 className="text-sm font-bold text-blue-800">{t('settings.backup')}</h3>
   648→      <p className="text-xs text-muted-foreground">{t('settings.backupDesc')}</p>
   649→      <Button
   650→        onClick={doBackup}
   651→        className="h-11 w-full bg-blue-600 text-white hover:bg-blue-700"
   652→        aria-label="Backup data"
   653→      >
   654→        <Download className="size-4" />
   655→        {t('settings.backupBtn')}
   656→      </Button>
   657→      <Button
   658→        variant="outline"
   659→        className="h-11 w-full"
   660→        onClick={() => fileRef.current?.click()}
   661→        aria-label="Restore backup"
   662→      >
   663→        <Upload className="size-4" />
   664→        {t('settings.restoreBtn')}
   665→      </Button>
   666→      <input
   667→        ref={fileRef}
   668→        type="file"
   669→        accept=".json,application/json"
   670→        className="hidden"
   671→        aria-label="Restore backup file"
   672→        onChange={onFile}
   673→      />
   674→
   675→      <AlertDialog open={!!pending} onOpenChange={(o) => !o && setPending(null)}>
   676→        <AlertDialogContent className="max-w-xs">
   677→          <AlertDialogHeader>
   678→            <AlertDialogTitle>{t('settings.restoreConfirmTitle')}</AlertDialogTitle>
   679→            <AlertDialogDescription>
   680→              {t('settings.restoreConfirmDesc')}
   681→              {pending && (
   682→                <span className="mt-1 block font-semibold text-foreground">
   683→                  {pending.settings.companyName} · {pending.products.length} {t('settings.backupProducts')} ·{' '}
   684→                  {(pending.invoices ?? []).length} {t('settings.backupBills')}
   685→                </span>
   686→              )}
   687→            </AlertDialogDescription>
   688→          </AlertDialogHeader>
   689→          <AlertDialogFooter>
   690→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   691→            <AlertDialogAction
   692→              onClick={doRestore}
   693→              className="bg-blue-600 text-white hover:bg-blue-700"
   694→              aria-label="Confirm restore"
   695→            >
   696→              {t('settings.restore')}
   697→            </AlertDialogAction>
   698→          </AlertDialogFooter>
   699→        </AlertDialogContent>
   700→      </AlertDialog>
   701→    </section>
   702→  );
   703→}

   705→/**
   706→ * Every device runs the SAME app link — when the owner (or we) change anything,
   707→ * installed phones pick it up automatically: the app checks for updates while
   708→ * open (and on every open/focus/back-online) and refreshes itself. This button
   709→ * lets the owner force the check right now.
   710→ */
   711→function AppUpdatesSection() {
   712→  const { toast } = useToast();
   713→  const [checking, setChecking] = useState(false);
   714→
   715→  const checkNow = async () => {
   716→    if (!('serviceWorker' in navigator)) {
   717→      toast({ title: 'You are on the latest version' });
   718→      return;
   719→    }
   720→    setChecking(true);
   721→    try {
   722→      const reg = await navigator.serviceWorker.getRegistration();
   723→      if (!reg) {
   724→        toast({ title: 'You are on the latest version' });
   725→        return;
   726→      }
   727→      let found = false;
   728→      const onUpdateFound = () => {
   729→        found = true;
   730→      };
   731→      reg.addEventListener('updatefound', onUpdateFound, { once: true });
   732→      await reg.update();
   733→      // give a freshly found worker a moment to install (skipWaiting applies it)
   734→      await new Promise((r) => setTimeout(r, 1200));
   735→      reg.removeEventListener('updatefound', onUpdateFound);
   736→      if (found || reg.waiting || reg.installing) {
   737→        toast({
   738→          title: 'Update found',
   739→          description: 'The app refreshes to the latest version in a moment…',
   740→        });
   741→      } else {
   742→        toast({ title: 'You are on the latest version' });
   743→      }
   744→    } catch {
   745→      toast({
   746→        title: 'Cannot check right now',
   747→        description: 'Go online once and tap again — the app will update itself.',
   748→      });
   749→    } finally {
   750→      setChecking(false);
   751→    }
   752→  };
   753→
   754→  return (
   755→    <section aria-label="App updates" className="space-y-2">
   756→      <h3 className="text-sm font-bold text-blue-800">App updates</h3>
   757→      <p className="text-xs text-muted-foreground">
   758→        The same app link works on every phone. When anything is updated here, all installed
   759→        devices get the change automatically — no reinstall needed. Bills stay safe on each
   760→        device.
   761→      </p>
   762→      <Button variant="outline" className="w-full" onClick={checkNow} disabled={checking}>
   763→        {checking ? <RefreshCw className="size-4 animate-spin" /> : <RefreshCw className="size-4" />}
   764→        Check for updates now
   765→      </Button>
   766→    </section>
   767→  );
   768→}

   770→function Field({
   771→  label,
   772→  value,
   773→  onChange,
   774→  placeholder,
   775→  type = 'text',
   776→  textarea,
   777→  ariaLabel,
   778→}: {
   779→  label: string;
   780→  value: string;
   781→  onChange: (v: string) => void;
   782→  placeholder?: string;
   783→  type?: string;
   784→  textarea?: boolean;
   785→  ariaLabel?: string;
   786→}) {
   787→  const isPhone = type === 'tel';
   788→  return (
   789→    <div className="space-y-1.5">
   790→      <Label className="text-xs text-muted-foreground">{label}</Label>
   791→      {textarea ? (
   792→        <textarea
   793→          aria-label={ariaLabel}
   794→          value={value}
   795→          onChange={(e) => onChange(e.target.value)}
   796→          placeholder={placeholder}
   797→          rows={2}
   798→          className="flex w-full rounded-md border border-input bg-transparent px-3 py-2 text-sm shadow-sm placeholder:text-muted-foreground focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
   799→        />
   800→      ) : (
   801→        <Input
   802→          aria-label={ariaLabel}
   803→          type={type}
   804→          inputMode={isPhone ? 'numeric' : undefined}
   805→          value={value}
   806→          onChange={(e) => onChange(e.target.value)}
   807→          onBlur={isPhone ? (e) => onChange(normalizePhone(e.target.value)) : undefined}
   808→          placeholder={placeholder}
   809→          className="h-11"
   810→        />
   811→      )}
   812→    </div>
   813→  );
   814→}
