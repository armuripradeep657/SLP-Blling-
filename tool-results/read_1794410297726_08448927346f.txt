   145→/** Bilingual labels — Telugu first, English second (used when bill language = Both) */
   146→export const BOTH_LABELS: SlipLabels = (() => {
   147→  const out = {} as SlipLabels;
   148→  (Object.keys(EN_LABELS) as (keyof SlipLabels)[]).forEach((k) => {
   149→    const en = EN_LABELS[k];
   150→    const te = TE_LABELS[k];
   151→    if (k === 'modes') {
   152→      out.modes = { Cash: 'నగదు / Cash', UPI: 'UPI', Card: 'కార్డ్ / Card', Due: 'బాకీ / Due' };
   153→    } else if (typeof en === 'string' && typeof te === 'string') {
   154→      (out as unknown as Record<string, string>)[k] = `${te} / ${en}`;
   155→    } else {
   156→      (out as unknown as Record<string, unknown>)[k] = en;
   157→    }
   158→  });
   159→  return out;
   160→})();
   161→
   162→export function labels(lang: BillLanguage | undefined): SlipLabels {
   163→  return lang === 'te' ? TE_LABELS : lang === 'both' ? BOTH_LABELS : EN_LABELS;
   164→}
   165→
   166→/** Currency prefix on bill amounts — Telugu bills use రూ., English/Both keep Rs. */
   167→export function cur(lang: BillLanguage | undefined): string {
   168→  return lang === 'te' ? 'రూ.' : 'Rs.';
   169→}
   170→
   171→/** Company name in the bill language — Telugu name when set, English otherwise */
   172→export function companyNameFor(
   173→  s: { companyName: string; companyNameTe?: string },
   174→  lang: BillLanguage | undefined
   175→): string {
   176→  return lang === 'te' && s.companyNameTe && s.companyNameTe.trim() ? s.companyNameTe : s.companyName;
   177→}
   178→
   179→/**
   180→ * Company name lines for printed documents — "Both" prints the Telugu name
   181→ * above the English name.
   182→ */
   183→export function companyLines(
   184→  s: { companyName: string; companyNameTe?: string },
   185→  lang: BillLanguage | undefined
   186→): string[] {
   187→  const te = (s.companyNameTe ?? '').trim();
   188→  const en = (s.companyName ?? '').trim();
   189→  if (lang === 'te') return [te || en];
   190→  if (lang === 'both') {
   191→    const lines = te && te !== en ? [te, en] : [en || te];
   192→    return lines.filter(Boolean);
   193→  }
   194→  return [en || te];
   195→}
   196→
   197→/** Item name in the bill language — Telugu product name when set, otherwise the English one */
   198→export function itemLabel(it: { name: string; nameTe?: string }, lang: BillLanguage | undefined): string {
   199→  const te = (it.nameTe ?? '').trim();
   200→  if (lang === 'te') return te || it.name;
   201→  if (lang === 'both') return te && te !== it.name ? `${te} / ${it.name}` : it.name;
   202→  return it.name;
   203→}
   204→
   205→/** Fill {tokens} in a label template */
   206→export function fill(template: string, vars: Record<string, string>): string {
   207→  return template.replace(/\{(\w+)\}/g, (_, k) => vars[k] ?? '');
   208→}
