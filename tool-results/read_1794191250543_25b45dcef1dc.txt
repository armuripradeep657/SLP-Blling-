    31→    31→const EXPENSE_CATEGORIES: ExpenseCategory[] = ['Labour wages', 'Diesel', 'Raw material', 'Other'];
    32→    32→
    33→    33→const CATEGORY_TE: Record<ExpenseCategory, string> = {
    34→    34→  'Labour wages': 'కూలీ వేతనాలు',
    35→    35→  Diesel: 'డీజిల్',

    41→    41→
    42→    42→type LedgerTab = 'day' | 'allDays' | 'dues' | 'expenses';
    43→    43→
    44→    44→/** One money row in the ledger — income (bill payment) or spent (expense) */
    45→    45→interface LedgerRow {
    46→    46→  time: string;
    47→    47→  /** main line — customer name for income, category for spent */
    48→    48→  label: string;
    49→    49→  /** secondary line — invoice no / note */
    50→    50→  sub: string;
    51→    51→  amount: number;
    52→    52→  kind: 'in' | 'out';
    53→    53→  /** Cash / UPI / Card — shown as a chip on every row */
    54→    54→  mode?: PaymentMode;
    55→    55→}
    56→    56→
    57→    57→function categoryLabel(c: ExpenseCategory, appLang: AppLanguage): string {
    58→    58→  return appLang === 'te' ? CATEGORY_TE[c] : c;
    59→    59→}
    60→    60→
    61→    61→function modeLabel(m: PaymentMode, appLang: AppLanguage): string {
    62→    62→  if (appLang !== 'te') return m;
    63→    63→  return m === 'Cash' ? 'నగదు' : m === 'Card' ? 'కార్డ్' : m === 'Due' ? 'బాకీ' : 'UPI';
    64→    64→}
    65→    65→
    66→    66→/** Cash / UPI / Card chip — on income and spent rows alike */
    67→    67→function ModeChip({ mode, appLang }: { mode?: PaymentMode; appLang: AppLanguage }) {
    68→    68→  if (!mode) return null;
    69→    69→  const cls =
    70→    70→    mode === 'Cash'
    71→    71→      ? 'bg-green-100 text-green-800'
    72→    72→      : mode === 'UPI'
    73→    73→        ? 'bg-blue-100 text-blue-800'
    74→    74→        : mode === 'Due'
    75→    75→          ? 'bg-amber-100 text-amber-800'
    76→    76→          : 'bg-purple-100 text-purple-800';
    77→    77→  return (
    78→    78→    <span className={`inline-flex shrink-0 items-center rounded-full px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wide ${cls}`}>
    79→    79→      {modeLabel(mode, appLang)}
    80→    80→    </span>
    81→    81→  );
    82→    82→}
    83→    83→
    84→    84→/** Stat card used across ledger tabs (module-level — never created during render) */
    85→    85→function StatCard({ label, value, kind }: { label: string; value: number; kind: 'in' | 'out' | 'net' }) {
    86→    86→  return (
    87→    87→    <div className="anim-pop rounded-xl border bg-card p-3">
    88→    88→      <p className="flex items-center gap-1 text-xs text-muted-foreground">
    89→    89→        {kind === 'in' ? (
    90→    90→          <ArrowUpRight className="size-3.5 text-green-700" />
    91→    91→        ) : kind === 'out' ? (
    92→    92→          <ArrowDownRight className="size-3.5 text-red-600" />
    93→    93→        ) : (
    94→    94→          <TrendingUp className="size-3.5 text-blue-700" />
    95→    95→        )}
    96→    96→        {label}
    97→    97→      </p>
    98→    98→      <p
    99→    99→        className={`mt-0.5 text-lg font-extrabold tabular-nums ${
   100→   100→          kind === 'in' ? 'text-green-700' : kind === 'out' ? 'text-red-600' : value < 0 ? 'text-red-600' : 'text-blue-800'
   101→   101→        }`}
   102→   102→      >
   103→   103→        ₹{fmt(value)}
   104→   104→      </p>
   105→   105→    </div>
   106→   106→  );
   107→   107→}
   108→   108→

   119→   119→            <ArrowUpRight className="size-4" />

   127→   127→          <p className="flex items-center gap-1.5 truncate text-sm font-medium">
   128→   128→            <span className="truncate">{r.label}</span>
   129→   129→            <ModeChip mode={r.mode} appLang={appLang} />
   130→   130→          </p>
   131→   131→          <p className="truncate text-[11px] text-muted-foreground">
   132→   132→            {r.time}
   133→   133→            {r.sub ? ` · ${r.sub}` : ''}
   134→   134→          </p>
   135→   135→        </div>

   137→   137→      <span className={`shrink-0 text-sm font-bold tabular-nums ${r.kind === 'in' ? 'text-green-700' : 'text-red-600'}`}>
   138→   138→        {r.kind === 'in' ? '+' : '−'} ₹{fmt(r.amount)}
   139→   139→      </span>
   140→   140→    </li>
   141→   141→  );
   142→   142→}
   143→   143→
   144→   144→export default function LedgerView() {
   145→   145→  const t = useT();
   146→   146→  const appLang = useAppLang();
   147→   147→  const invoices = useStore((s) => s.invoices);
   148→   148→  const expenses = useStore((s) => s.expenses);
   149→   149→  const receiveDuePayment = useStore((s) => s.receiveDuePayment);
   150→   150→  const addExpense = useStore((s) => s.addExpense);
   151→   151→  const deleteExpense = useStore((s) => s.deleteExpense);
   152→   152→  const { toast } = useToast();
   153→   153→
   154→   154→  const today = localDateKey(new Date());
   155→   155→  const thisMonth = localMonthKey(new Date());
   156→   156→
   157→   157→  const [tab, setTab] = useState<LedgerTab>('day');
   158→   158→  const [day, setDay] = useState(today);
   159→   159→  const [month, setMonth] = useState(thisMonth);
   160→   160→  const [expandedDay, setExpandedDay] = useState<string | null>(null);
   161→   161→
   162→   162→  /* ---------------- Money aggregation ---------------- */
   163→   163→
   164→   164→  // Every income / spent entry grouped by local day — feeds the day view,
   165→   165→  // the all-days list AND each day's expandable detail (name + Cash/UPI)
   166→   166→  const rowsByDay = useMemo(() => {
   167→   167→    const map = new Map<string, LedgerRow[]>();
   168→   168→    const push = (k: string, r: LedgerRow) => {
   169→   169→      const list = map.get(k);
   170→   170→      if (list) list.push(r);
   171→   171→      else map.set(k, [r]);
   172→   172→    };
   173→   173→    invoices.forEach((inv) => {
   174→   174→      paymentsOf(inv).forEach((p) => {
   175→   175→        push(localDateKey(new Date(p.dateISO)), {
   176→   176→          time: new Date(p.dateISO).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' }),
   177→   177→          label: `${t('ledger.receivedFrom')} ${inv.customer.name || inv.no}`,
   178→   178→          sub: inv.no,
   179→   179→          amount: p.amount,
   180→   180→          kind: 'in',
   181→   181→          mode: p.mode,
   182→   182→        });
   183→   183→      });
   184→   184→    });
   185→   185→    expenses.forEach((e) => {
   186→   186→      push(localDateKey(new Date(e.dateISO)), {
   187→   187→        time: new Date(e.dateISO).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' }),
   188→   188→        label: categoryLabel(e.category, appLang),
   189→   189→        sub: e.note || '',
   190→   190→        amount: e.amount,
   191→   191→        kind: 'out',
   192→   192→        mode: e.mode,
   193→   193→      });
   194→   194→    });
   195→   195→    // newest first within each day
   196→   196→    map.forEach((list, k) => map.set(k, [...list].sort((a, b) => b.time.localeCompare(a.time))));
   197→   197→    return map;
   198→   198→  }, [invoices, expenses, t, appLang]);
   199→   199→
   200→   200→  const dayRows = rowsByDay.get(day) ?? [];
   201→   201→
   202→   202→  const moneyByDay = useMemo(() => {
   203→   203→    const map = new Map<string, { income: number; spent: number }>();
   204→   204→    rowsByDay.forEach((rows, k) => {
   205→   205→      const acc = { income: 0, spent: 0 };
   206→   206→      rows.forEach((r) => (r.kind === 'in' ? (acc.income += r.amount) : (acc.spent += r.amount)));
   207→   207→      map.set(k, { income: Math.round(acc.income * 100) / 100, spent: Math.round(acc.spent * 100) / 100 });
   208→   208→    });
   209→   209→    return map;
   210→   210→  }, [rowsByDay]);
   211→   211→
   212→   212→  const dayMoney = (k: string): { income: number; spent: number } =>
   213→   213→    moneyByDay.get(k) ?? { income: 0, spent: 0 };
   214→   214→
   215→   215→  const dues = useMemo(
   216→   216→    () => invoices.filter((i) => i.totals.balance > 0).sort((a, b) => b.dateISO.localeCompare(a.dateISO)),
   217→   217→    [invoices]
   218→   218→  );
   219→   219→  const totalDue = dues.reduce((s, i) => s + i.totals.balance, 0);
   220→   220→
   221→   221→  // all-days rows for the selected month
   222→   222→  const monthDays = useMemo(
   223→   223→    () =>
   224→   224→      [...moneyByDay.keys()]
   225→   225→        .filter((k) => localMonthKey(new Date(`${k}T12:00:00`)) === month)
   226→   226→        .sort((a, b) => b.localeCompare(a))
   227→   227→        .map((k) => ({ key: k, ...(moneyByDay.get(k) as { income: number; spent: number }) })),
   228→   228→    [moneyByDay, month]
   229→   229→  );
   230→   230→
   231→   231→  const monthIncome = monthDays.reduce((s, d) => s + d.income, 0);
   232→   232→  const monthSpent = monthDays.reduce((s, d) => s + d.spent, 0);
   233→   233→
   234→   234→  const monthExpenses = useMemo(
   235→   235→    () =>
   236→   236→      expenses
   237→   237→        .filter((e) => localMonthKey(new Date(e.dateISO)) === month)
   238→   238→        .sort((a, b) => b.dateISO.localeCompare(a.dateISO)),
   239→   239→    [expenses, month]
   240→   240→  );
   241→   241→
   242→   242→  /* ---------------- Receive-due inline panel ---------------- */
   243→   243→  const [receiving, setReceiving] = useState<string | null>(null);
   244→   244→  const [recvAmount, setRecvAmount] = useState('');
   245→   245→  const [recvMode, setRecvMode] = useState<PaymentMode>('Cash');
   246→   246→
   247→   247→  const openReceive = (id: string, balance: number) => {
   248→   248→    setReceiving(id);
   249→   249→    setRecvAmount(String(fmtShort(balance)));
   250→   250→    setRecvMode('Cash');
   251→   251→  };
   252→   252→
   253→   253→  const confirmReceive = () => {
   254→   254→    if (!receiving) return;
   255→   255→    const amt = Number(recvAmount) || 0;
   256→   256→    if (amt <= 0) return;
   257→   257→    receiveDuePayment(receiving, amt, recvMode);
   258→   258→    setReceiving(null);
   259→   259→    toast({ title: t('ledger.receiveDone'), description: `₹${fmt(amt)} · ${modeLabel(recvMode, appLang)}` });
   260→   260→  };
   261→   261→
   262→   262→  /* ---------------- Expense form ---------------- */
   263→   263→  const [expDate, setExpDate] = useState(today);
   264→   264→  const [expCat, setExpCat] = useState<ExpenseCategory>('Labour wages');
   265→   265→  const [expAmount, setExpAmount] = useState('');
   266→   266→  const [expNote, setExpNote] = useState('');
   267→   267→  const [expMode, setExpMode] = useState<PaymentMode>('Cash');
   268→   268→
   269→   269→  const submitExpense = () => {
   270→   270→    const amt = Number(expAmount) || 0;
   271→   271→    if (amt <= 0) return;
   272→   272→    // date input gives YYYY-MM-DD — keep it local (noon ISO to avoid TZ shifts)
   273→   273→    addExpense({
   274→   274→      dateISO: new Date(`${expDate}T12:00:00`).toISOString(),
   275→   275→      category: expCat,
   276→   276→      amount: amt,
   277→   277→      note: expNote.trim() || undefined,
   278→   278→      mode: expMode,
   279→   279→    });
   280→   280→    setExpAmount('');
   281→   281→    setExpNote('');
   282→   282→    toast({
   283→   283→      title: t('ledger.expAdded'),
   284→   284→      description: `₹${fmt(amt)} · ${categoryLabel(expCat, appLang)} · ${modeLabel(expMode, appLang)}`,
   285→   285→    });
   286→   286→  };
   287→   287→
   288→   288→  const tabs: { id: LedgerTab; label: string }[] = [
   289→   289→    { id: 'day', label: t('ledger.day') },
   290→   290→    { id: 'allDays', label: t('ledger.allDays') },
   291→   291→    { id: 'dues', label: `${t('ledger.dues')}${totalDue > 0 ? ` (${dues.length})` : ''}` },
   292→   292→    { id: 'expenses', label: t('ledger.expenses') },
   293→   293→  ];
   294→   294→
   295→   295→  return (
   296→   296→    <div className="space-y-3">
   297→   297→      {/* Tab chips */}
   298→   298→      <div className="flex gap-1.5 overflow-x-auto pb-1" role="tablist" aria-label="Ledger sections">
   299→   299→        {tabs.map((x) => (
   300→   300→          <button
   301→   301→            key={x.id}
   302→   302→            role="tab"
   303→   303→            aria-selected={tab === x.id}
   304→   304→            onClick={() => setTab(x.id)}
   305→   305→            className={`shrink-0 rounded-full px-3.5 py-1.5 text-xs font-medium transition-all active:scale-95 ${
   306→   306→              tab === x.id ? 'bg-blue-600 text-white' : 'border bg-card hover:bg-accent'
   307→   307→            }`}
   308→   308→          >
   309→   309→            {x.label}
   310→   310→          </button>
   311→   311→        ))}
   312→   312→      </div>
   313→   313→
   314→   314→      {/* ---------- DAY VIEW ---------- */}
   315→   315→      {tab === 'day' && (
   316→   316→        <div className="space-y-3">
   317→   317→          <div className="flex items-center gap-2">
   318→   318→            <CalendarDays className="size-4 shrink-0 text-blue-700" />
   319→   319→            <Input type="date" value={day} max={today} onChange={(e) => setDay(e.target.value)} className="h-11 max-w-44" aria-label="Ledger day" />
   320→   320→            {day !== today && (
   321→   321→              <Button variant="ghost" size="sm" className="h-9" onClick={() => setDay(today)}>
   322→   322→                {appLang === 'te' ? 'ఈ రోజు' : 'Today'}
   323→   323→              </Button>
   324→   324→            )}
   325→   325→          </div>
   326→   326→          <div className="grid grid-cols-3 gap-2">
   327→   327→            <StatCard label={t('ledger.income')} value={dayMoney(day).income} kind="in" />
   328→   328→            <StatCard label={t('ledger.spent')} value={dayMoney(day).spent} kind="out" />
   329→   329→            <StatCard label={t('ledger.net')} value={dayMoney(day).income - dayMoney(day).spent} kind="net" />
   330→   330→          </div>
   331→   331→          <ul className="space-y-1.5">
   332→   332→            {dayRows.map((r, i) => (
   333→   333→              <LedgerRowItem key={`${r.time}-${r.label}-${i}`} r={r} appLang={appLang} i={i} />
   334→   334→            ))}
   335→   335→            {dayRows.length === 0 && (
   336→   336→              <li className="py-10 text-center text-sm text-muted-foreground">{t('ledger.noActivity')}</li>
   337→   337→            )}
   338→   338→          </ul>
   339→   339→        </div>
   340→   340→      )}
   341→   341→
   342→   342→      {/* ---------- ALL DAYS BY MONTH (tap a day for its details) ---------- */}
   343→   343→      {tab === 'allDays' && (
   344→   344→        <div className="space-y-3">
   345→   345→          <div className="flex items-center gap-2">
   346→   346→            <CalendarDays className="size-4 shrink-0 text-blue-700" />
   347→   347→            <Input type="month" value={month} onChange={(e) => setMonth(e.target.value || thisMonth)} className="h-11 max-w-44" aria-label="Ledger month" />
   348→   348→          </div>
   349→   349→          <div className="grid grid-cols-3 gap-2">
   350→   350→            <StatCard label={t('ledger.income')} value={monthIncome} kind="in" />
   351→   351→            <StatCard label={t('ledger.spent')} value={monthSpent} kind="out" />
   352→   352→            <StatCard label={t('ledger.net')} value={monthIncome - monthSpent} kind="net" />
   353→   353→          </div>
   354→   354→          <div className="overflow-hidden rounded-xl border bg-card">
   355→   355→            <div className="grid grid-cols-[1fr_auto_auto_auto] gap-x-3 bg-blue-600 px-3 py-2 text-[11px] font-bold text-white">
   356→   356→              <span>{t('ledger.date')}</span>
   357→   357→              <span className="w-20 text-right">{t('ledger.income')}</span>
   358→   358→              <span className="w-20 text-right">{t('ledger.spent')}</span>
   359→   359→              <span className="w-20 text-right">{t('ledger.net')}</span>
   360→   360→            </div>
   361→   361→            <ul className="max-h-96 divide-y overflow-y-auto">
   362→   362→              {monthDays.map((d) => {
   363→   363→                const net = Math.round((d.income - d.spent) * 100) / 100;
   364→   364→                const dt = new Date(`${d.key}T12:00:00`);
   365→   365→                const open = expandedDay === d.key;
   366→   366→                const rows = rowsByDay.get(d.key) ?? [];
   367→   367→                return (
   368→   368→                  <li key={d.key} className={open ? 'bg-blue-50/50' : undefined}>
   369→   369→                    <button
   370→   370→                      onClick={() => setExpandedDay(open ? null : d.key)}
   371→   371→                      aria-expanded={open}
   372→   372→                      className="grid w-full grid-cols-[1fr_auto_auto_auto] gap-x-3 px-3 py-2 text-xs tabular-nums transition-colors hover:bg-accent/60"
   373→   373→                    >
   374→   374→                      <span className="flex items-center gap-1 font-medium">
   375→   375→                        <ChevronDown className={`size-3.5 shrink-0 text-muted-foreground transition-transform ${open ? 'rotate-180' : ''}`} />
   376→   376→                        {dt.toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: '2-digit' })}
   377→   377→                      </span>
   378→   378→                      <span className="w-20 text-right font-semibold text-green-700">₹{fmtShort(d.income)}</span>
   379→   379→                      <span className="w-20 text-right font-semibold text-red-600">₹{fmtShort(d.spent)}</span>
   380→   380→                      <span className={`w-20 text-right font-bold ${net < 0 ? 'text-red-600' : 'text-blue-800'}`}>₹{fmtShort(net)}</span>
   381→   381→                    </button>
   382→   382→                    {/* Day detail — every income & spent entry with name + Cash/UPI */}
   383→   383→                    {open && (
   384→   384→                      <div className="anim-fade-up space-y-1 px-2.5 pb-2.5">
   385→   385→                        {rows.map((r, i) => (
   386→   386→                          <div key={i} className="flex items-center justify-between gap-2 rounded-lg border bg-card px-2.5 py-1.5">
   387→   387→                            <div className="flex min-w-0 items-center gap-1.5">
   388→   388→                              {r.kind === 'in' ? (
   389→   389→                                <ArrowUpRight className="size-3.5 shrink-0 text-green-700" />
   390→   390→                              ) : (
   391→   391→                                <ArrowDownRight className="size-3.5 shrink-0 text-red-600" />
