    38→    38→
    39→    39→function modeLabel(m: PaymentMode, appLang: AppLanguage): string {
    40→    40→  if (appLang !== 'te') return m;
    41→    41→  return m === 'Cash' ? 'నగదు' : m === 'Card' ? 'కార్డ్' : m === 'Due' ? 'బాకీ' : 'UPI';
    42→    42→}
    43→    43→
    44→    44→/** Cash / UPI / Card chip — how the money was given that day */
    45→    45→function ModeChip({ mode, appLang }: { mode?: PaymentMode; appLang: AppLanguage }) {
    46→    46→  if (!mode) return null;
    47→    47→  const cls =
    48→    48→    mode === 'Cash'
    49→    49→      ? 'bg-green-100 text-green-800'
    50→    50→      : mode === 'UPI'
    51→    51→        ? 'bg-blue-100 text-blue-800'
    52→    52→        : 'bg-purple-100 text-purple-800';
    53→    53→  return (
    54→    54→    <span
    55→    55→      className={`inline-flex shrink-0 items-center rounded-full px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wide ${cls}`}
    56→    56→    >
    57→    57→      {modeLabel(mode, appLang)}
    58→    58→    </span>
    59→    59→  );
    60→    60→}
