    43→    43→
    44→    44→/** One money row in the ledger — income (bill payment) or spent (expense) */
    45→    45→interface LedgerRow {
    46→    46→  time: string;
    47→    47→  /** main line — customer name for income, category for spent */
    48→    48→  label: string;
    49→    49→  /** secondary line — invoice no / note */
    50→    50→  sub: string;
    51→    51→  amount: number;
    52→    52→  kind: 'in' | 'out';
    53→    53→  /** Cash / UPI / Card — shown as a chip on every row */

    55→    55→}
    56→    56→
    57→    57→function categoryLabel(c: ExpenseCategory, appLang: AppLanguage): string {
    58→    58→  return appLang === 'te' ? CATEGORY_TE[c] : c;
    59→    59→}
    60→    60→
    61→    61→function modeLabel(m: PaymentMode, appLang: AppLanguage): string {
    62→    62→  if (appLang !== 'te') return m;
    63→    63→  return m === 'Cash' ? 'నగదు' : m === 'Card' ? 'కార్డ్' : m === 'Due' ? 'బాకీ' : 'UPI';
    64→    64→}
    65→    65→
    66→    66→/** Cash / UPI / Card chip — on income and spent rows alike */
    67→    67→function ModeChip({ mode, appLang }: { mode?: PaymentMode; appLang: AppLanguage }) {
    68→    68→  if (!mode) return null;
    69→    69→  const cls =
    70→    70→    mode === 'Cash'
    71→    71→      ? 'bg-green-100 text-green-800'
    72→    72→      : mode === 'UPI'
    73→    73→        ? 'bg-blue-100 text-blue-800'
    74→    74→        : mode === 'Due'
    75→    75→          ? 'bg-amber-100 text-amber-800'
    76→    76→          : 'bg-purple-100 text-purple-800';
    77→    77→  return (
    78→    78→    <span className={`inline-flex shrink-0 items-center rounded-full px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wide ${cls}`}>
    79→    79→      {modeLabel(mode, appLang)}
    80→    80→    </span>
    81→    81→  );
    82→    82→}
    83→    83→
    84→    84→/** Stat card used across ledger tabs (module-level — never created during render) */

    87→    87→    <div className="anim-pop rounded-xl border bg-card p-3">
    88→    88→      <p className="flex items-center gap-1 text-xs text-muted-foreground">
    89→    89→        {kind === 'in' ? (
    90→    90→          <ArrowUpRight className="size-3.5 text-green-700" />
    91→    91→        ) : kind === 'out' ? (
    92→    92→          <ArrowDownRight className="size-3.5 text-red-600" />
    93→    93→        ) : (
    94→    94→          <TrendingUp className="size-3.5 text-blue-700" />

    98→    98→      <p
    99→    99→        className={`mt-0.5 text-lg font-extrabold tabular-nums ${
   100→   100→          kind === 'in' ? 'text-green-700' : kind === 'out' ? 'text-red-600' : value < 0 ? 'text-red-600' : 'text-blue-800'
   101→   101→        }`}
   102→   102→      >
   103→   103→        ₹{fmt(value)}
   104→   104→      </p>
   105→   105→    </div>
   106→   106→  );
   107→   107→}

   143→   143→
   144→   144→export default function LedgerView() {
   145→   145→  const t = useT();
   146→   146→  const appLang = useAppLang();
   147→   147→  const invoices = useStore((s) => s.invoices);
   148→   148→  const expenses = useStore((s) => s.expenses);
   149→   149→  const incomes = useStore((s) => s.incomes);
   150→   150→  const receiveDuePayment = useStore((s) => s.receiveDuePayment);
   151→   151→  const addExpense = useStore((s) => s.addExpense);
   152→   152→  const deleteExpense = useStore((s) => s.deleteExpense);
   153→   153→  const addIncome = useStore((s) => s.addIncome);
   154→   154→  const deleteIncome = useStore((s) => s.deleteIncome);
   155→   155→  const { toast } = useToast();
   156→   156→
   157→   157→  const today = localDateKey(new Date());
   158→   158→  const thisMonth = localMonthKey(new Date());
   159→   159→
   160→   160→  const [tab, setTab] = useState<LedgerTab>('day');
   161→   161→  const [day, setDay] = useState(today);
   162→   162→  const [month, setMonth] = useState(thisMonth);
   163→   163→  const [expandedDay, setExpandedDay] = useState<string | null>(null);
   164→   164→
   165→   165→  /* ---------------- Money aggregation ---------------- */
   166→   166→
   167→   167→  // Every income / spent entry grouped by local day — feeds the day view,
   168→   168→  // the all-days list AND each day's expandable detail (name + Cash/UPI)
   169→   169→  const rowsByDay = useMemo(() => {
   170→   170→    const map = new Map<string, LedgerRow[]>();
   171→   171→    const push = (k: string, r: LedgerRow) => {
   172→   172→      const list = map.get(k);
   173→   173→      if (list) list.push(r);
   174→   174→      else map.set(k, [r]);
   175→   175→    };
   176→   176→    invoices.forEach((inv) => {
   177→   177→      paymentsOf(inv).forEach((p) => {
   178→   178→        push(localDateKey(new Date(p.dateISO)), {
   179→   179→          time: new Date(p.dateISO).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' }),
   180→   180→          label: `${t('ledger.receivedFrom')} ${inv.customer.name || inv.no}`,
   181→   181→          sub: inv.no,
   182→   182→          amount: p.amount,
   183→   183→          kind: 'in',
   184→   184→          mode: p.mode,
   185→   185→        });
   186→   186→      });
   187→   187→    });
   188→   188→    expenses.forEach((e) => {
   189→   189→      push(localDateKey(new Date(e.dateISO)), {
   190→   190→        time: new Date(e.dateISO).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' }),
   191→   191→        label: categoryLabel(e.category, appLang),
   192→   192→        sub: e.note || '',
   193→   193→        amount: e.amount,
   194→   194→        kind: 'out',
   195→   195→        mode: e.mode,
   196→   196→      });
   197→   197→    });
   198→   198→    // manual income entries — money received outside a bill, with its own name
   199→   199→    incomes.forEach((e) => {
   200→   200→      push(localDateKey(new Date(e.dateISO)), {
   201→   201→        time: new Date(e.dateISO).toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' }),
   202→   202→        label: e.name,
   203→   203→        sub: '',
   204→   204→        amount: e.amount,
   205→   205→        kind: 'in',
   206→   206→        mode: e.mode,
   207→   207→      });
   208→   208→    });
   209→   209→    // newest first within each day
   210→   210→    map.forEach((list, k) => map.set(k, [...list].sort((a, b) => b.time.localeCompare(a.time))));
   211→   211→    return map;
   212→   212→  }, [invoices, expenses, incomes, t, appLang]);
   213→   213→
   214→   214→  const dayRows = rowsByDay.get(day) ?? [];
   215→   215→
   216→   216→  const moneyByDay = useMemo(() => {
   217→   217→    const map = new Map<string, { income: number; spent: number }>();
   218→   218→    rowsByDay.forEach((rows, k) => {
   219→   219→      const acc = { income: 0, spent: 0 };
   220→   220→      rows.forEach((r) => (r.kind === 'in' ? (acc.income += r.amount) : (acc.spent += r.amount)));
   221→   221→      map.set(k, { income: Math.round(acc.income * 100) / 100, spent: Math.round(acc.spent * 100) / 100 });
   222→   222→    });
   223→   223→    return map;
   224→   224→  }, [rowsByDay]);
   225→   225→
   226→   226→  const dayMoney = (k: string): { income: number; spent: number } =>
   227→   227→    moneyByDay.get(k) ?? { income: 0, spent: 0 };
   228→   228→
   229→   229→  const dues = useMemo(
   230→   230→    () => invoices.filter((i) => i.totals.balance > 0).sort((a, b) => b.dateISO.localeCompare(a.dateISO)),
   231→   231→    [invoices]
   232→   232→  );
   233→   233→  const totalDue = dues.reduce((s, i) => s + i.totals.balance, 0);
   234→   234→
   235→   235→  // all-days rows for the selected month
   236→   236→  const monthDays = useMemo(
   237→   237→    () =>
   238→   238→      [...moneyByDay.keys()]
   239→   239→        .filter((k) => localMonthKey(new Date(`${k}T12:00:00`)) === month)
   240→   240→        .sort((a, b) => b.localeCompare(a))
   241→   241→        .map((k) => ({ key: k, ...(moneyByDay.get(k) as { income: number; spent: number }) })),
   242→   242→    [moneyByDay, month]
   243→   243→  );
   244→   244→
   245→   245→  const monthIncome = monthDays.reduce((s, d) => s + d.income, 0);
   246→   246→  const monthSpent = monthDays.reduce((s, d) => s + d.spent, 0);
   247→   247→
   248→   248→  const monthExpenses = useMemo(
   249→   249→    () =>
   250→   250→      expenses
   251→   251→        .filter((e) => localMonthKey(new Date(e.dateISO)) === month)
   252→   252→        .sort((a, b) => b.dateISO.localeCompare(a.dateISO)),
   253→   253→    [expenses, month]
   254→   254→  );
   255→   255→
   256→   256→  const monthIncomes = useMemo(
   257→   257→    () =>
   258→   258→      incomes
   259→   259→        .filter((e) => localMonthKey(new Date(e.dateISO)) === month)
   260→   260→        .sort((a, b) => b.dateISO.localeCompare(a.dateISO)),
   261→   261→    [incomes, month]
   262→   262→  );
   263→   263→
   264→   264→  /* ---------------- Receive-due inline panel ---------------- */
   265→   265→  const [receiving, setReceiving] = useState<string | null>(null);
   266→   266→  const [recvAmount, setRecvAmount] = useState('');
   267→   267→  const [recvMode, setRecvMode] = useState<PaymentMode>('Cash');
   268→   268→
   269→   269→  const openReceive = (id: string, balance: number) => {
   270→   270→    setReceiving(id);
   271→   271→    setRecvAmount(String(fmtShort(balance)));
   272→   272→    setRecvMode('Cash');
   273→   273→  };
   274→   274→
   275→   275→  const confirmReceive = () => {
   276→   276→    if (!receiving) return;
   277→   277→    const amt = Number(recvAmount) || 0;
   278→   278→    if (amt <= 0) return;
   279→   279→    receiveDuePayment(receiving, amt, recvMode);
   280→   280→    setReceiving(null);
   281→   281→    toast({ title: t('ledger.receiveDone'), description: `₹${fmt(amt)} · ${modeLabel(recvMode, appLang)}` });
   282→   282→  };
   283→   283→
   284→   284→  /* ---------------- Expense form ---------------- */
   285→   285→  const [expDate, setExpDate] = useState(today);
   286→   286→  const [expCat, setExpCat] = useState<ExpenseCategory>('Labour wages');
   287→   287→  const [expAmount, setExpAmount] = useState('');
   288→   288→  const [expNote, setExpNote] = useState('');
   289→   289→  const [expMode, setExpMode] = useState<PaymentMode>('Cash');
   290→   290→
   291→   291→  const submitExpense = () => {
   292→   292→    const amt = Number(expAmount) || 0;
   293→   293→    if (amt <= 0) return;
   294→   294→    // date input gives YYYY-MM-DD — keep it local (noon ISO to avoid TZ shifts)
   295→   295→    addExpense({
   296→   296→      dateISO: new Date(`${expDate}T12:00:00`).toISOString(),
   297→   297→      category: expCat,
   298→   298→      amount: amt,
   299→   299→      note: expNote.trim() || undefined,
   300→   300→      mode: expMode,
   301→   301→    });
   302→   302→    setExpAmount('');
   303→   303→    setExpNote('');
   304→   304→    toast({
   305→   305→      title: t('ledger.expAdded'),
   306→   306→      description: `₹${fmt(amt)} · ${categoryLabel(expCat, appLang)} · ${modeLabel(expMode, appLang)}`,
   307→   307→    });
   308→   308→  };
   309→   309→
   310→   310→  /* ---------------- Manual income form ---------------- */
   311→   311→  const [incDate, setIncDate] = useState(today);
   312→   312→  const [incName, setIncName] = useState('');
   313→   313→  const [incAmount, setIncAmount] = useState('');
   314→   314→  const [incMode, setIncMode] = useState<PaymentMode>('Cash');
   315→   315→
   316→   316→  const submitIncome = () => {
   317→   317→    const amt = Number(incAmount) || 0;
   318→   318→    if (amt <= 0 || !incName.trim()) return;
   319→   319→    addIncome({
   320→   320→      dateISO: new Date(`${incDate}T12:00:00`).toISOString(),
   321→   321→      name: incName.trim(),
   322→   322→      amount: amt,
   323→   323→      mode: incMode,
   324→   324→    });
   325→   325→    setIncName('');
   326→   326→    setIncAmount('');
   327→   327→    toast({
   328→   328→      title: t('ledger.incAdded'),
   329→   329→      description: `₹${fmt(amt)} · ${incName.trim()} · ${modeLabel(incMode, appLang)}`,
   330→   330→    });
   331→   331→  };
   332→   332→
   333→   333→  const tabs: { id: LedgerTab; label: string }[] = [
   334→   334→    { id: 'day', label: t('ledger.day') },
   335→   335→    { id: 'allDays', label: t('ledger.allDays') },
   336→   336→    { id: 'income', label: t('ledger.income') },
   337→   337→    { id: 'expenses', label: t('ledger.expenses') },
   338→   338→    { id: 'dues', label: `${t('ledger.dues')}${totalDue > 0 ? ` (${dues.length})` : ''}` },
   339→   339→  ];
   340→   340→
   341→   341→  return (
   342→   342→    <div className="space-y-3">

   347→   347→            key={x.id}
   348→   348→            role="tab"
   349→   349→            aria-selected={tab === x.id}
   350→   350→            onClick={() => setTab(x.id)}
   351→   351→            className={`shrink-0 rounded-full px-3.5 py-1.5 text-xs font-medium transition-all active:scale-95 ${

   355→   355→            {x.label}

   357→   357→        ))}
   358→   358→      </div>
   359→   359→
   360→   360→      {/* ---------- DAY VIEW ---------- */}

   383→   383→            )}
   384→   384→          </ul>
   385→   385→        </div>
   386→   386→      )}
   387→   387→
   388→   388→      {/* ---------- ALL DAYS BY MONTH (tap a day for its details) ---------- */}
   389→   389→      {tab === 'allDays' && (
   390→   390→        <div className="space-y-3">
   391→   391→          <div className="flex items-center gap-2">
   392→   392→            <CalendarDays className="size-4 shrink-0 text-blue-700" />
   393→   393→            <Input type="month" value={month} onChange={(e) => setMonth(e.target.value || thisMonth)} className="h-11 max-w-44" aria-label="Ledger month" />

   395→   395→          <div className="grid grid-cols-3 gap-2">
   396→   396→            <StatCard label={t('ledger.income')} value={monthIncome} kind="in" />
   397→   397→            <StatCard label={t('ledger.spent')} value={monthSpent} kind="out" />
   398→   398→            <StatCard label={t('ledger.net')} value={monthIncome - monthSpent} kind="net" />
   399→   399→          </div>
   400→   400→          <div className="overflow-hidden rounded-xl border bg-card">
   401→   401→            <div className="grid grid-cols-[1fr_auto_auto_auto] gap-x-3 bg-blue-600 px-3 py-2 text-[11px] font-bold text-white">
   402→   402→              <span>{t('ledger.date')}</span>
   403→   403→              <span className="w-20 text-right">{t('ledger.income')}</span>
   404→   404→              <span className="w-20 text-right">{t('ledger.spent')}</span>
